// ============================================================

var SHEET_OPS = 'OperacionesV2';
var SHEET_MHE_ID = '10GQdpwjUMCQqEGGDCalWJV5FAupzJSyW8dULY9JDWbc';
var SHEET_MHE_NAME_1 = 'SABANA_MAESTRA';
var SHEET_MHE_NAME_2 = 'SÁBANA_MAESTRA';

// Columnas de la hoja OperacionesV2 (índice 0 = col A)
var C = {
  UID:0, FECHA:1, ANALISTA:2, PRODUCTO:3, TIPO_ID:4,
  IDENTIFICADOR:5, ESTADO:6, TIER:7, OBS:8,
  ASIGNADO_EN:9, COMPLETADO_EN:10, CONFIRMADO_POR:11,
  HORAS_HAB:12, ID_OP:13, NEEDS_REVIEW:14
};
var NCOLS = 15;
var HEADERS = ['UID','Fecha','Analista','Producto','TipoID','Identificador',
               'Estado','Tier','Observacion','AsignadoEn','CompletatadoEn',
               'ConfirmadoPor','HorasHabiles','IDOperacion','NeedsReview'];

function doPost(e) {
  try {
    var params = JSON.parse(e.postData.contents);
    var result = handleRequest(params);
    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// ── Obtener / crear hoja ─────────────────────────────────────
function getOpsSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_OPS);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_OPS);
    sheet.getRange(1,1,1,NCOLS).setValues([HEADERS]);
    sheet.getRange(1,1,1,NCOLS).setBackground('#002960').setFontColor('#FFFFFF').setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// ── Construir mapa UID → {row, data} ─────────────────────────
function buildUIDMap(sheet) {
  var data = sheet.getDataRange().getValues();
  var map = {};
  for (var i = 1; i < data.length; i++) {
    if (data[i][C.UID]) map[data[i][C.UID]] = {row: i+1, data: data[i]};
  }
  return {map: map, data: data, total: data.length};
}

// ── Convertir fila de Sheets a objeto op ─────────────────────
function rowToOp(row) {
  return {
    uid:            row[C.UID] || '',
    fecha:          row[C.FECHA] instanceof Date ? Utilities.formatDate(row[C.FECHA],'UTC','yyyy-MM-dd') : (row[C.FECHA]||'').toString().slice(0,10),
    analista:       row[C.ANALISTA] || '',
    productName:    row[C.PRODUCTO] || '',
    idType:         row[C.TIPO_ID] || '',
    identificador:  row[C.IDENTIFICADOR] || '',
    estado:         row[C.ESTADO] || 'PENDIENTE',
    tier:           row[C.TIER] || 'normal',
    obs_seguimiento:row[C.OBS] || '',
    asignado_en:    row[C.ASIGNADO_EN] instanceof Date ? row[C.ASIGNADO_EN].toISOString() : (row[C.ASIGNADO_EN]||''),
    completado_en:  row[C.COMPLETADO_EN] instanceof Date ? row[C.COMPLETADO_EN].toISOString() : (row[C.COMPLETADO_EN]||''),
    confirmedBy:    row[C.CONFIRMADO_POR] || '',
    _horas_habiles: parseFloat(row[C.HORAS_HAB])||0,
    _id_operacion:  row[C.ID_OP] || '',
    _needs_review:  row[C.NEEDS_REVIEW] === true || row[C.NEEDS_REVIEW] === 'TRUE'
  };
}

// ── Convertir objeto op a fila de Sheets ─────────────────────
function opToRow(op, date, confirmedBy) {
  var row = new Array(NCOLS).fill('');
  row[C.UID]           = op.uid || '';
  row[C.FECHA]         = date || op.fecha || '';
  row[C.ANALISTA]      = op.analista || '';
  row[C.PRODUCTO]      = op.productName || op.productKey || '';
  row[C.TIPO_ID]       = op.idType || '';
  row[C.IDENTIFICADOR] = op.identificador || '';
  row[C.ESTADO]        = op.estado || 'PENDIENTE';
  row[C.TIER]          = op.tier || 'normal';
  row[C.OBS]           = op.obs_seguimiento || '';
  row[C.ASIGNADO_EN]   = op.asignado_en ? new Date(op.asignado_en) : '';
  row[C.COMPLETADO_EN] = op.completado_en ? new Date(op.completado_en) : '';
  row[C.CONFIRMADO_POR]= confirmedBy || op.confirmedBy || '';
  row[C.HORAS_HAB]     = op._horas_habiles || 0;
  row[C.ID_OP]         = op._id_operacion || '';
  row[C.NEEDS_REVIEW]  = op._needs_review ? 'TRUE' : 'FALSE';
  return row;
}

// ── HANDLE REQUEST ────────────────────────────────────────────
function handleRequest(params) {
  var action = params.action;

  // ── ping ──────────────────────────────────────────────────
  if (action === 'ping') return {ok: true, version: '2.0'};

  // ── autoMigrate: migrar ops de GestionOps a OperacionesV2 si faltan ──────
  if (action === 'autoMigrate') {
    var date = params.date;
    var sheet = getOpsSheet();
    var result = buildUIDMap(sheet);
    // Contar ops ya en OperacionesV2 para esta fecha
    var existing = 0;
    for (var i = 1; i < result.data.length; i++) {
      var rowDate = result.data[i][C.FECHA] instanceof Date
        ? Utilities.formatDate(result.data[i][C.FECHA],'UTC','yyyy-MM-dd')
        : (result.data[i][C.FECHA]||'').toString().slice(0,10);
      if (rowDate === date) existing++;
    }
    // Si ya hay ops para esa fecha, no migrar
    if (existing > 0) return {migrated: 0, existing: existing};
    // Buscar en GestionOps
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var legacySheet = ss.getSheetByName('GestionOps');
    if (!legacySheet) return {migrated: 0, error: 'No existe GestionOps'};
    var legacyData = legacySheet.getDataRange().getValues();
    for (var j = 1; j < legacyData.length; j++) {
      if (legacyData[j][0] === 'ops:' + date && legacyData[j][1]) {
        try {
          var ops = JSON.parse(legacyData[j][1]);
          if (!ops || !ops.length) return {migrated: 0};
          var rows = ops.map(function(op) {
            return opToRow(op, date, op.confirmedBy||'');
          });
          sheet.getRange(result.total+1, 1, rows.length, NCOLS).setValues(rows);
          return {migrated: rows.length, date: date};
        } catch(e) {
          return {migrated: 0, error: e.toString()};
        }
      }
    }
    return {migrated: 0, notFound: true};
  }

  // ── getHistorial: obtener días únicos con resumen ─────────
  if (action === 'getHistorial') {
    var sheet = getOpsSheet();
    var result = buildUIDMap(sheet);
    var days = {};
    for (var i = 1; i < result.data.length; i++) {
      var row = result.data[i];
      if (!row[C.UID]) continue;
      var rowDate = row[C.FECHA] instanceof Date
        ? Utilities.formatDate(row[C.FECHA],'UTC','yyyy-MM-dd')
        : (row[C.FECHA]||'').toString().slice(0,10);
      if (!rowDate) continue;
      if (!days[rowDate]) days[rowDate] = {date:rowDate, total:0, hechos:0, confirmedBy:''};
      days[rowDate].total++;
      if ((row[C.ESTADO]||'') === 'HECHO') days[rowDate].hechos++;
      if (row[C.CONFIRMADO_POR] && !days[rowDate].confirmedBy) days[rowDate].confirmedBy = row[C.CONFIRMADO_POR];
    }
    var dayList = Object.values(days).sort(function(a,b){ return b.date.localeCompare(a.date); });
    return {days: dayList};
  }

  // ── getOps: leer todas las ops de una fecha ───────────────
  if (action === 'getOps') {
    var date = params.date;
    var sheet = getOpsSheet();
    var result = buildUIDMap(sheet);
    var ops = [];
    for (var i = 1; i < result.data.length; i++) {
      var row = result.data[i];
      if (!row[C.UID]) continue;
      var rowDate = row[C.FECHA] instanceof Date
        ? Utilities.formatDate(row[C.FECHA],'UTC','yyyy-MM-dd')
        : (row[C.FECHA]||'').toString().slice(0,10);
      if (rowDate === date) ops.push(rowToOp(row));
    }
    // Fallback: si OperacionesV2 no tiene datos para esta fecha,
    // leer desde GestionOps (JSON blob v1) y migrar automaticamente
    if (ops.length === 0) {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var legacySheet = ss.getSheetByName('GestionOps');
      if (legacySheet) {
        var legacyData = legacySheet.getDataRange().getValues();
        for (var lj = 1; lj < legacyData.length; lj++) {
          if (legacyData[lj][0] === 'ops:' + date && legacyData[lj][1]) {
            try {
              var legacyOps = JSON.parse(legacyData[lj][1]);
              if (legacyOps && legacyOps.length > 0) {
                // Migrar a OperacionesV2 automaticamente
                var newRows = legacyOps.map(function(op){ return opToRow(op, date, ''); });
                sheet.getRange(result.total+1, 1, newRows.length, NCOLS).setValues(newRows);
                return {ops: legacyOps, total: legacyOps.length, migrated: true};
              }
            } catch(e) {}
          }
        }
      }
    }
    return {ops: ops, total: ops.length};
  }

  // ── setOps: confirmar reparto (solo inserta nuevas, no toca existentes) ──
  if (action === 'setOps') {
    var sheet = getOpsSheet();
    var res = buildUIDMap(sheet);
    var ops = params.ops || [];
    var date = params.date;
    var confirmedBy = params.confirmedBy || '';
    var newRows = [];
    var updated = 0;

    for (var j = 0; j < ops.length; j++) {
      var op = ops[j];
      if (res.map[op.uid]) {
        // Ya existe — NO tocar si tiene HECHO, solo actualizar identificador si cambió
        var existing = res.map[op.uid].data;
        var newId = op.identificador || '';
        if (existing[C.ESTADO] !== 'HECHO' && newId && newId !== existing[C.IDENTIFICADOR]) {
          sheet.getRange(res.map[op.uid].row, C.IDENTIFICADOR+1).setValue(newId);
        }
        updated++;
      } else {
        // Nueva — agregar al batch
        newRows.push(opToRow(op, date, confirmedBy));
      }
    }
    // Insertar todas las nuevas en una sola operación (no fila por fila)
    if (newRows.length > 0) {
      var lastRow = sheet.getLastRow();
      sheet.getRange(lastRow+1, 1, newRows.length, NCOLS).setValues(newRows);
    }
    return {success: true, inserted: newRows.length, updated: updated};
  }

  // ── markEstado: marcar 1 op HECHO/PENDIENTE (ultra rápido) ─
  if (action === 'markEstado') {
    var sheet = getOpsSheet();
    var res = buildUIDMap(sheet);
    var uid = params.uid;
    if (res.map[uid]) {
      // v2.0: actualizar fila en OperacionesV2
      var rowNum = res.map[uid].row;
      sheet.getRange(rowNum, C.ESTADO+1).setValue(params.estado || 'PENDIENTE');
      sheet.getRange(rowNum, C.COMPLETADO_EN+1).setValue(
        params.completado_en ? new Date(params.completado_en) : ''
      );
      return {success: true};
    } else {
      // Fallback v1.0: actualizar JSON blob en GestionOps
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var legacySheet = ss.getSheetByName('GestionOps');
      if (!legacySheet) return {error: 'UID no encontrado: ' + uid};
      var legacyData = legacySheet.getDataRange().getValues();
      for (var k = 1; k < legacyData.length; k++) {
        if (legacyData[k][0] && legacyData[k][0].toString().startsWith('ops:') && legacyData[k][1]) {
          try {
            var ops = JSON.parse(legacyData[k][1]);
            var opIdx = ops.findIndex(function(o){ return o.uid === uid; });
            if (opIdx >= 0) {
              ops[opIdx].estado = params.estado || 'PENDIENTE';
              ops[opIdx].completado_en = params.completado_en || null;
              legacySheet.getRange(k+1, 2).setValue(JSON.stringify(ops));
              legacySheet.getRange(k+1, 3).setValue(new Date().toISOString());
              return {success: true, legacy: true};
            }
          } catch(e) {}
        }
      }
      return {error: 'UID no encontrado en v1 ni v2: ' + uid};
    }
  }

  // ── updateField: actualizar 1 campo de 1 op ───────────────
  if (action === 'updateField') {
    var sheet = getOpsSheet();
    var res = buildUIDMap(sheet);
    if (!res.map[params.uid]) return {error: 'UID no encontrado'};
    var colMap = {
      'identificador': C.IDENTIFICADOR,
      'analista': C.ANALISTA,
      'obs_seguimiento': C.OBS,
      'estado': C.ESTADO,
      'tier': C.TIER
    };
    var col = colMap[params.field];
    if (col === undefined) return {error: 'Campo desconocido: ' + params.field};
    sheet.getRange(res.map[params.uid].row, col+1).setValue(params.value || '');
    return {success: true};
  }

  // ── getMHEPendientes: leer SÁBANA_MAESTRA (sin cambios) ───
  if (action === 'getMHEPendientes') {
    try {
      var masterSS = SpreadsheetApp.openById(SHEET_MHE_ID);
      var masterSheet = masterSS.getSheetByName(SHEET_MHE_NAME_1) || masterSS.getSheetByName(SHEET_MHE_NAME_2);
      if (!masterSheet) return {error: 'No se encontro la pestana SABANA_MAESTRA'};
      // Leer solo columnas necesarias para optimizar velocidad
      var lastRow = masterSheet.getLastRow();
      var lastCol = masterSheet.getLastColumn();
      var headers = masterSheet.getRange(1, 1, 1, lastCol).getValues()[0];
      var colIdx = {};
      headers.forEach(function(h, i){ colIdx[h] = i; });
      var required = ['ID_Operacion','RUT_Titular','Segmento_SLA','Resolucion_Riesgo','Horas_Habiles','Fecha_Envio','Analista_Riesgo'];
      for (var r = 0; r < required.length; r++) {
        if (colIdx[required[r]] === undefined) return {error: 'Columna no encontrada: ' + required[r]};
      }
      // Leer solo las columnas necesarias (no todo el sheet)
      var needCols = ['ID_Operacion','RUT_Titular','Segmento_SLA','Resolucion_Riesgo','Horas_Habiles',
                      'Fecha_Envio','Analista_Riesgo','Categoria_SLA','derivar_operacion_IVAN'];
      var colNums = needCols.map(function(c){ return colIdx[c] !== undefined ? colIdx[c]+1 : null; });
      // Leer todas las filas en bloques de 2000 para evitar timeout
      var ops = [];
      var BLOCK = 2000;
      for (var start = 2; start <= lastRow; start += BLOCK) {
        var end = Math.min(start + BLOCK - 1, lastRow);
        var block = masterSheet.getRange(start, 1, end - start + 1, lastCol).getValues();
        for (var i = 0; i < block.length; i++) {
          var row = block[i];
          // Filtro 1: Pendiente
          var resolucion = (row[colIdx['Resolucion_Riesgo']] || '').toString().trim();
          if (resolucion !== 'Pendiente') continue;
          // Filtro 2: En_Agente_IA = TRUE
          var en_agente = (row[colIdx['En_Agente_IA']] || '').toString().trim().toUpperCase();
          if (en_agente !== 'TRUE') continue;
          // Filtro 3: oferta_valida_deudor = TRUE
          var oferta_d = (row[colIdx['oferta_valida_deudor']] || '').toString().trim().toUpperCase();
          if (oferta_d !== 'TRUE') continue;
          // Filtro 4: oferta_valida_codeudor = TRUE o vacío
          var oferta_c = (row[colIdx['oferta_valida_codeudor']] || '').toString().trim().toUpperCase();
          if (oferta_c !== 'TRUE' && oferta_c !== '' && oferta_c !== 'NO_APLICA') continue;
          // Filtro 5: paso_por_ia = TRUE
          var paso_ia = (row[colIdx['paso_por_ia']] || '').toString().trim().toUpperCase();
          if (paso_ia !== 'TRUE') continue;
          // Filtro 6: derivar_operacion_IVAN = TRUE
          var derivar = (row[colIdx['derivar_operacion_IVAN']] || '').toString().trim().toUpperCase();
          if (derivar !== 'TRUE') continue;
          // Pasó todos los filtros — agregar
          var segmento = (row[colIdx['Segmento_SLA']] || '').toString().trim();
          var horas = parseFloat(row[colIdx['Horas_Habiles']]) || 0;
          var tier = horas >= 48 ? 'vencido' : (horas >= 36 ? 'urgente' : 'normal');
          ops.push({
            id_operacion:   (row[colIdx['ID_Operacion']] || '').toString(),
            rut:            (row[colIdx['RUT_Titular']] || '').toString(),
            segmento:       segmento,
            horas_habiles:  horas,
            tier:           tier,
            fecha_envio:    (row[colIdx['Fecha_Envio']] || '').toString(),
            analista_actual:(row[colIdx['Analista_Riesgo']] || '').toString(),
            categoria_sla:  (row[colIdx['Categoria_SLA']] || 'Inmediata').toString(),
            derivar_ivan:   true
          });
        }
      }
      var tierOrder = {vencido:0, urgente:1, normal:2};
      ops.sort(function(a,b){ return tierOrder[a.tier] - tierOrder[b.tier]; });
      return {ops: ops, total: ops.length};
    } catch(err) {
      return {error: err.toString()};
    }
  }

  // ── config:team (compatibilidad con v1) ───────────────────
  if (action === 'set') {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('GestionOps') || ss.insertSheet('GestionOps');
    if (sheet.getLastRow() === 0) sheet.appendRow(['Clave','Datos','Actualizado']);
    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === params.key) {
        sheet.getRange(i+1,2).setValue(params.value);
        sheet.getRange(i+1,3).setValue(new Date().toISOString());
        return {success:true};
      }
    }
    sheet.appendRow([params.key, params.value, new Date().toISOString()]);
    return {success:true};
  }
  if (action === 'get') {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('GestionOps');
    if (!sheet) return {key:params.key, value:null};
    var data = sheet.getDataRange().getValues();
    for (var j = 1; j < data.length; j++) {
      if (data[j][0] === params.key) return {key:params.key, value:data[j][1]};
    }
    return {key:params.key, value:null};
  }
  if (action === 'list') {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('GestionOps');
    if (!sheet) return {keys:[]};
    var prefix = params.prefix || '';
    var data = sheet.getDataRange().getValues();
    var keys = [];
    for (var k = 1; k < data.length; k++) {
      if (data[k][0] && data[k][0].toString().startsWith(prefix)) keys.push(data[k][0].toString());
    }
    return {keys:keys};
  }

  // ── setMeta: guardar planificación del día ───────────────
  if (action === 'setMeta') {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var metaSheet = ss.getSheetByName('PlanificacionMeta');
    if (!metaSheet) {
      metaSheet = ss.insertSheet('PlanificacionMeta');
      metaSheet.appendRow(['Fecha','DayType','TotalOps','ConfirmadoPor','ConfirmadoEn',
        'Vol_Leasing','Vol_Pie','Vol_PV','Vol_Subsidio','Vol_Brasil','Distribucion']);
      metaSheet.getRange(1,1,1,11).setBackground('#002960').setFontColor('#FFFFFF').setFontWeight('bold');
      metaSheet.setFrozenRows(1);
    }
    var meta = params.meta;
    var vols = meta.volumes || {};
    var rowArr = [
      new Date(meta.date),
      meta.dayType || '',
      meta.totalOps || 0,
      meta.confirmedBy || '',
      meta.confirmedAt ? new Date(meta.confirmedAt) : '',
      (vols.leasing && vols.leasing.total) || 0,
      (vols.pie && vols.pie.total) || 0,
      (vols.pv && vols.pv.total) || 0,
      (vols.subsidio && vols.subsidio.total) || 0,
      (vols.brasil && vols.brasil.total) || 0,
      JSON.stringify(meta.distribution || {})
    ];
    // Buscar si ya existe fila para esta fecha y actualizar
    var metaData = metaSheet.getDataRange().getValues();
    var dateStr = params.date;
    var found = false;
    for (var mi = 1; mi < metaData.length; mi++) {
      var rowDate = metaData[mi][0] instanceof Date
        ? Utilities.formatDate(metaData[mi][0],'UTC','yyyy-MM-dd')
        : (metaData[mi][0]||'').toString().slice(0,10);
      if (rowDate === dateStr) {
        metaSheet.getRange(mi+1, 1, 1, rowArr.length).setValues([rowArr]);
        found = true;
        break;
      }
    }
    if (!found) metaSheet.appendRow(rowArr);
    return {success: true};
  }

  // ── getMeta: obtener planificación de un día ──────────────
  if (action === 'getMeta') {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var metaSheet = ss.getSheetByName('PlanificacionMeta');
    if (!metaSheet) return {meta: null};
    var metaData = metaSheet.getDataRange().getValues();
    var dateStr = params.date;
    for (var mj = 1; mj < metaData.length; mj++) {
      var rowDate = metaData[mj][0] instanceof Date
        ? Utilities.formatDate(metaData[mj][0],'UTC','yyyy-MM-dd')
        : (metaData[mj][0]||'').toString().slice(0,10);
      if (rowDate === dateStr) {
        return {meta: {
          date: dateStr,
          dayType: metaData[mj][1] || '',
          totalOps: metaData[mj][2] || 0,
          confirmedBy: metaData[mj][3] || '',
          confirmedAt: metaData[mj][4] || '',
          volumes: {
            leasing: {total: metaData[mj][5]||0},
            pie: {total: metaData[mj][6]||0},
            pv: {total: metaData[mj][7]||0},
            subsidio: {total: metaData[mj][8]||0},
            brasil: {total: metaData[mj][9]||0}
          },
          distribution: JSON.parse(metaData[mj][10]||'{}')
        }};
      }
    }
    return {meta: null};
  }

  return {error: 'Acción no reconocida: ' + action};
}
